CREATE DATABASE IF NOT EXISTS inventoryoli;
USE inventoryoli;

-- ========================
-- TABLE ROLES
-- ========================
CREATE TABLE roles (
    role_id INT AUTO_INCREMENT PRIMARY KEY,
    nama_role VARCHAR(100) NOT NULL UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- ========================
-- TABLE USERS
-- ========================
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    nama VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL,
    password VARCHAR(255) NOT NULL,
    role_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_users_roles
        FOREIGN KEY (role_id)
        REFERENCES roles(role_id)
);

-- ========================
-- TABLE KATEGORI
-- ========================
CREATE TABLE kategori (
    kategori_id INT AUTO_INCREMENT PRIMARY KEY,
    nama_kategori VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- ========================
-- TABLE PEMASOK
-- ========================
CREATE TABLE pemasok (
    pemasok_id INT AUTO_INCREMENT PRIMARY KEY,
    nama_pemasok VARCHAR(255) NOT NULL,
    alamat VARCHAR(255) NOT NULL,
    kontak VARCHAR(20) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- ========================
-- TABLE PRODUK
-- ========================
CREATE TABLE produk (
    produk_id INT AUTO_INCREMENT PRIMARY KEY,
    kode_barang INT NOT NULL,
    nama_barang VARCHAR(255) NOT NULL,
    stok INT NOT NULL,
    stok_minimum INT NOT NULL,
    harga_beli INT NOT NULL,
    harga_jual INT NOT NULL,
    kategori_id INT NOT NULL,
    pemasok_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT produk_ibfk_1
        FOREIGN KEY (kategori_id)
        REFERENCES kategori(kategori_id),

    CONSTRAINT produk_ibfk_2
        FOREIGN KEY (pemasok_id)
        REFERENCES pemasok(pemasok_id)
);

-- ========================
-- TABLE PELANGGAN
-- ========================
CREATE TABLE pelanggan (
    pelanggan_id INT AUTO_INCREMENT PRIMARY KEY,
    nama_bengkel VARCHAR(255) NOT NULL,
    no_telp VARCHAR(20) NOT NULL,
    alamat VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- ========================
-- TABLE BARANG KELUAR
-- ========================
CREATE TABLE barang_keluar (
    barang_keluar_id INT AUTO_INCREMENT PRIMARY KEY,
    tanggal DATETIME DEFAULT CURRENT_TIMESTAMP,
    jumlah INT NOT NULL,
    harga_jual INT NOT NULL,
    total INT NOT NULL,
    produk_id INT NOT NULL,
    user_id INT NOT NULL,
    pelanggan_id INT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT barang_keluar_ibfk_1
        FOREIGN KEY (user_id)
        REFERENCES users(user_id),

    CONSTRAINT barang_keluar_ibfk_2
        FOREIGN KEY (produk_id)
        REFERENCES produk(produk_id),

    CONSTRAINT barang_keluar_ibfk_3
        FOREIGN KEY (pelanggan_id)
        REFERENCES pelanggan(pelanggan_id)
);

-- ========================
-- TABLE BARANG MASUK
-- ========================
CREATE TABLE barang_masuk (
    barang_masuk_id INT AUTO_INCREMENT PRIMARY KEY,
    tanggal DATETIME DEFAULT CURRENT_TIMESTAMP,
    jumlah INT NOT NULL,
    harga_beli INT NOT NULL,
    total INT NOT NULL,
    user_id INT NOT NULL,
    produk_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT barang_masuk_ibfk_1
        FOREIGN KEY (user_id)
        REFERENCES users(user_id),

    CONSTRAINT barang_masuk_ibfk_2
        FOREIGN KEY (produk_id)
        REFERENCES produk(produk_id)
);

-- ========================
-- TABLE HUTANG PEMBELIAN
-- ========================
CREATE TABLE hutang_pembelian (
    hutang_id INT AUTO_INCREMENT PRIMARY KEY,
    barang_masuk_id INT NOT NULL,
    total_hutang INT NOT NULL,
    sisa_hutang INT NOT NULL,
    jumlah_bayar INT DEFAULT 0,
    total_terbayar INT DEFAULT 0,
    tanggal_transaksi DATETIME DEFAULT CURRENT_TIMESTAMP,
    tanggal_jatuh_tempo DATE,
    tanggal_bayar_terakhir DATE,
    status VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT hutang_pembelian_ibfk_1
        FOREIGN KEY (barang_masuk_id)
        REFERENCES barang_masuk(barang_masuk_id)
);

-- ========================
-- TABLE LOG AKTIVITAS
-- ========================
CREATE TABLE log_aktivitas (
    log_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    aktivitas VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT log_aktivitas_ibfk_1
        FOREIGN KEY (user_id)
        REFERENCES users(user_id)
);

-- ========================
-- TABLE MIGRATIONS
-- ========================
CREATE TABLE migrations (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    migration VARCHAR(255) NOT NULL,
    batch INT NOT NULL
);

-- ========================
-- TABLE PEMASUKAN
-- ========================
CREATE TABLE pemasukan (
    pemasukan_id INT AUTO_INCREMENT PRIMARY KEY,
    jumlah INT NOT NULL,
    tanggal DATETIME DEFAULT CURRENT_TIMESTAMP,
    keterangan VARCHAR(255) NOT NULL,
    barang_keluar_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pemasukan_ibfk_1
        FOREIGN KEY (barang_keluar_id)
        REFERENCES barang_keluar(barang_keluar_id)
);

-- ========================
-- TABLE PENGELUARAN
-- ========================
CREATE TABLE pengeluaran (
    pengeluaran_id INT AUTO_INCREMENT PRIMARY KEY,
    jumlah INT NOT NULL,
    tanggal DATETIME DEFAULT CURRENT_TIMESTAMP,
    keterangan VARCHAR(255) NOT NULL,
    barang_masuk_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pengeluaran_ibfk_1
        FOREIGN KEY (barang_masuk_id)
        REFERENCES barang_masuk(barang_masuk_id)
);

-- ========================
-- TABLE PIUTANG PELANGGAN
-- ========================
CREATE TABLE piutang_pelanggan (
    piutang_id INT AUTO_INCREMENT PRIMARY KEY,
    barang_keluar_id INT NOT NULL,
    total_piutang INT NOT NULL,
    sisa_piutang INT NOT NULL,
    jumlah_bayar INT DEFAULT 0,
    total_terbayar INT DEFAULT 0,
    tanggal_transaksi DATETIME DEFAULT CURRENT_TIMESTAMP,
    tanggal_jatuh_tempo DATE,
    tanggal_bayar_terakhir DATE,
    status VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT piutang_pelanggan_ibfk_1
        FOREIGN KEY (barang_keluar_id)
        REFERENCES barang_keluar(barang_keluar_id)
);
